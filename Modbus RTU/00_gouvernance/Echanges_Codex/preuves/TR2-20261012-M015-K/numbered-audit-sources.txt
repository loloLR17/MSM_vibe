0133 void iis3dwb_read_data_polling(void)
0134 {
0135   stmdev_ctx_t dev_ctx;
0136
0137   /* Initialize mems driver interface */
0138   dev_ctx.write_reg = platform_write;
0139   dev_ctx.read_reg = platform_read;
0140   dev_ctx.mdelay = platform_delay;
0141   dev_ctx.handle = &SENSOR_BUS;
0142
0143   /* Init test platform */
0144   platform_init();
0145
0146   /* Wait sensor boot time */
0147   platform_delay(BOOT_TIME);
0148
0149   /* Check device ID */
0150   iis3dwb_device_id_get(&dev_ctx, &whoamI);
0151
0152   if (whoamI != IIS3DWB_ID)
0153     while (1);
0154
0155   /* Restore default configuration */
0156   iis3dwb_reset_set(&dev_ctx, PROPERTY_ENABLE);
0157
0158   do {
0159     iis3dwb_reset_get(&dev_ctx, &rst);
0160   } while (rst);
0161
0162   /* Enable Block Data Update */
0163   iis3dwb_block_data_update_set(&dev_ctx, PROPERTY_ENABLE);
0164
0165   /* Set Output Data Rate */
0166   iis3dwb_xl_data_rate_set(&dev_ctx, IIS3DWB_XL_ODR_26k7Hz);
0167
0168   /* Set full scale */
0169   iis3dwb_xl_full_scale_set(&dev_ctx, IIS3DWB_2g);
0170
0171   /* Configure filtering chain(No aux interface)
0172    * Accelerometer low pass filter path
0173    */
0174   iis3dwb_xl_filt_path_on_out_set(&dev_ctx, IIS3DWB_LP_ODR_DIV_100);
0175
0176   /* Read samples in polling mode (no int) */
0177   while (1) {
0178     uint8_t reg;
0179
0180     /* Read output only if new xl value is available */
0181     iis3dwb_xl_flag_data_ready_get(&dev_ctx, &reg);
0182
0183     if (reg) {
0184       /* Read acceleration field data */
0185       memset(data_raw_acceleration, 0x00, 3 * sizeof(int16_t));
0186       iis3dwb_acceleration_raw_get(&dev_ctx, data_raw_acceleration);
0187       acceleration_mg[0] =
0188         iis3dwb_from_fs2g_to_mg(data_raw_acceleration[0]);
0189       acceleration_mg[1] =
0190         iis3dwb_from_fs2g_to_mg(data_raw_acceleration[1]);
0191       acceleration_mg[2] =
0192         iis3dwb_from_fs2g_to_mg(data_raw_acceleration[2]);
0193       snprintf((char *)tx_buffer, sizeof(tx_buffer),
0194               "Acceleration [mg]:%4.2f\t%4.2f\t%4.2f\r\n",
0195               acceleration_mg[0], acceleration_mg[1], acceleration_mg[2]);
0196       tx_com(tx_buffer, strlen((char const *)tx_buffer));
0197     }
0198
0199     iis3dwb_temp_flag_data_ready_get(&dev_ctx, &reg);
0200
0201     if (reg) {
0202       /* Read temperature data */
0203       memset(&data_raw_temperature, 0x00, sizeof(int16_t));
0204       iis3dwb_temperature_raw_get(&dev_ctx, &data_raw_temperature);
0205       temperature_degC = iis3dwb_from_lsb_to_celsius(data_raw_temperature);
0206       snprintf((char *)tx_buffer, sizeof(tx_buffer),
0207               "Temperature [degC]:%6.2f\r\n", temperature_degC);
0208       tx_com(tx_buffer, strlen((char const *)tx_buffer));
0209     }
0210   }
0211 }

0201

0204

0207           continue;
0208
0209         f_data = &fifo_data[k];
0210
0211         /* Read FIFO sensor value */

0213         datay = (int16_t *)&f_data->data[2];
0214         dataz = (int16_t *)&f_data->data[4];

0216

0223                   iis3dwb_from_fs8g_to_mg(*dataz));
0224           tx_com(tx_buffer, strlen((char const *)tx_buffer));

0228           tx_com(tx_buffer, strlen((char const *)tx_buffer));
0229           break;
0230         default:
0231           break;
0232         }
0233       }
0234
0235       snprintf((char *)tx_buffer, sizeof(tx_buffer), "------ \r\n\r\n");
0236       tx_com(tx_buffer, strlen((char const *)tx_buffer));
0237     }
0238   }
