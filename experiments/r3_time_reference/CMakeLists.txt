cmake_minimum_required(VERSION 3.20)
project(r3_time_reference LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
set(DURABLE "${CMAKE_CURRENT_SOURCE_DIR}/../r3_durable_fat32/src")
option(TIME_HOST_TESTS "Only frame and reference arithmetic tests" OFF)
if(TIME_HOST_TESTS)
 enable_testing()
 add_executable(frame_fixture tests/frame_fixture.c src/reference_frame.c "${DURABLE}/archive_format.c")
 target_include_directories(frame_fixture PRIVATE src "${DURABLE}")
 target_compile_options(frame_fixture PRIVATE -Wall -Wextra -Werror)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_test(NAME reference_bounds COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference.py" $<TARGET_FILE:frame_fixture>)
 return()
endif()
if(NOT CMAKE_CROSSCOMPILING)
 message(FATAL_ERROR "Qualified ARM toolchain or TIME_HOST_TESTS required")
endif()
set(STM32CUBE_U5_ROOT "" CACHE PATH "Qualified CubeU5 v1.9.0")
set(HAL "${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver")
set(DEV "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx")
set(SDK)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex)
 list(APPEND SDK "${HAL}/Src/stm32u5xx_${module}.c")
endforeach()
add_executable(r3_time_probe src/main.c src/reference_frame.c "${DURABLE}/archive_format.c" "${CMAKE_CURRENT_SOURCE_DIR}/../gpio_dat123_st_reference/src/clock.c" ${SDK} "${DEV}/Source/Templates/gcc/startup_stm32u575xx.s" "${DEV}/Source/Templates/system_stm32u5xx.c")
target_include_directories(r3_time_probe PRIVATE src "${DURABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../iis3dwb_st_reference/src" "${HAL}/Inc" "${DEV}/Include" "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include")
target_compile_definitions(r3_time_probe PRIVATE STM32U575xx USE_HAL_DRIVER)
target_compile_options(r3_time_probe PRIVATE -mcpu=cortex-m33 -mthumb -O2 -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections -fstack-usage)
get_filename_component(FW "${CMAKE_CURRENT_SOURCE_DIR}/../../Modbus RTU/05_Firmware" ABSOLUTE)
target_link_options(r3_time_probe PRIVATE -mcpu=cortex-m33 -mthumb --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/r3_time_probe.map" "-T${FW}/platform/stm32/linker/STM32U575ZITXQ_FLASH.ld")
set_target_properties(r3_time_probe PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET r3_time_probe POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:r3_time_probe> ${CMAKE_CURRENT_BINARY_DIR}/r3_time_probe.bin COMMAND ${SIZE} $<TARGET_FILE:r3_time_probe>)
