cmake_minimum_required(VERSION 3.20)
project(gpio_dat123_st_reference LANGUAGES C ASM)
if(NOT CMAKE_CROSSCOMPILING)
 message(FATAL_ERROR "Use cmake/arm-none-eabi-gcc.cmake")
endif()
set(STM32CUBE_U5_ROOT "" CACHE PATH "Recursive STM32CubeU5 v1.9.0")
set(HAL ${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver)
set(DEV ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx)
set(SOURCES)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex)
 set(source ${HAL}/Src/stm32u5xx_${module}.c)
 if(NOT EXISTS ${source})
  message(FATAL_ERROR "Missing SDK source: ${source}")
 endif()
 list(APPEND SOURCES ${source})
endforeach()
add_executable(gpio_dat123 src/main.c src/clock.c ${SOURCES}
 ${DEV}/Source/Templates/gcc/startup_stm32u575xx.s
 ${DEV}/Source/Templates/system_stm32u5xx.c)
target_include_directories(gpio_dat123 PRIVATE src ${HAL}/Inc ${DEV}/Include ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include)
target_compile_definitions(gpio_dat123 PRIVATE STM32U575xx USE_HAL_DRIVER)
target_compile_options(gpio_dat123 PRIVATE -mcpu=cortex-m33 -mthumb -O2 -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_link_options(gpio_dat123 PRIVATE -mcpu=cortex-m33 -mthumb --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/gpio_dat123.map -T${CMAKE_CURRENT_SOURCE_DIR}/STM32U575ZITXQ_FLASH.ld)
set_target_properties(gpio_dat123 PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET gpio_dat123 POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:gpio_dat123> ${CMAKE_CURRENT_BINARY_DIR}/gpio_dat123.bin COMMAND ${SIZE} $<TARGET_FILE:gpio_dat123>)
