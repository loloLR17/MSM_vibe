cmake_minimum_required(VERSION 3.20)
project(r2_spi2_sd LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
option(R2_HOST_TESTS "Offline SDHC model only, no hardware proof" OFF)
if(R2_HOST_TESTS)
 enable_testing()
 add_executable(test_sd_spi tests/test_sd_spi.c src/sd_spi.c)
 target_include_directories(test_sd_spi PRIVATE src)
 target_compile_options(test_sd_spi PRIVATE -O2 -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
 add_test(NAME sdhc_spi_protocol_and_faults_model COMMAND test_sd_spi)
 return()
endif()
if(NOT CMAKE_CROSSCOMPILING)
 message(FATAL_ERROR "Use qualified ARM toolchain or R2_HOST_TESTS")
endif()
get_filename_component(FW "${CMAKE_CURRENT_SOURCE_DIR}/../../Modbus RTU/05_Firmware" ABSOLUTE)
set(STM32CUBE_U5_ROOT "" CACHE PATH "Qualified CubeU5 v1.9.0")
set(HAL "${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver")
set(DEV "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx")
set(SDK)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex)
 if(NOT EXISTS "${HAL}/Src/stm32u5xx_${module}.c")
  message(FATAL_ERROR "Missing CubeU5 ${module}")
 endif()
 list(APPEND SDK "${HAL}/Src/stm32u5xx_${module}.c")
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OBSERVED "${CMAKE_CURRENT_BINARY_DIR}/generated/stm32u5xx_hal_spi_observed.c")
execute_process(COMMAND ${Python3_EXECUTABLE} -B "${CMAKE_CURRENT_SOURCE_DIR}/../iis3dwb_st_reference/tools/instrument_hal.py" "${HAL}/Src/stm32u5xx_hal_spi.c" "${OBSERVED}" RESULT_VARIABLE OBS_RESULT)
if(NOT OBS_RESULT EQUAL 0)
 message(FATAL_ERROR "Qualified CubeU5 SPI fingerprint refused")
endif()
add_executable(r2_spi2_sd src/main.c src/sd_spi.c "${CMAKE_CURRENT_SOURCE_DIR}/../gpio_dat123_st_reference/src/clock.c" "${OBSERVED}" ${SDK}
 "${DEV}/Source/Templates/gcc/startup_stm32u575xx.s" "${DEV}/Source/Templates/system_stm32u5xx.c")
target_include_directories(r2_spi2_sd PRIVATE src "${CMAKE_CURRENT_SOURCE_DIR}/../iis3dwb_st_reference/src" "${HAL}/Inc" "${DEV}/Include" "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include")
target_compile_definitions(r2_spi2_sd PRIVATE STM32U575xx USE_HAL_DRIVER)
target_compile_options(r2_spi2_sd PRIVATE -mcpu=cortex-m33 -mthumb -O2 -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections -fstack-usage)
target_link_options(r2_spi2_sd PRIVATE -mcpu=cortex-m33 -mthumb --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/r2_spi2_sd.map" "-T${FW}/platform/stm32/linker/STM32U575ZITXQ_FLASH.ld")
set_target_properties(r2_spi2_sd PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET r2_spi2_sd POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:r2_spi2_sd> ${CMAKE_CURRENT_BINARY_DIR}/r2_spi2_sd.bin COMMAND ${SIZE} $<TARGET_FILE:r2_spi2_sd>)
