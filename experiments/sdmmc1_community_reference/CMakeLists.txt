cmake_minimum_required(VERSION 3.20)
project(sdmmc1_community_reference LANGUAGES C ASM)
set(REFERENCE_ROOT "" CACHE PATH "Unchanged checkout VILLO88/NUCLEOU575 at 1e3c9183bad023ebf7d234134823d9270a0e8b64")
find_package(Git REQUIRED)
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${REFERENCE_ROOT} rev-parse HEAD OUTPUT_VARIABLE REF_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT REF_HEAD STREQUAL "1e3c9183bad023ebf7d234134823d9270a0e8b64")
 message(FATAL_ERROR "Wrong reference commit")
endif()
execute_process(COMMAND ${GIT_EXECUTABLE} -C ${REFERENCE_ROOT} status --porcelain OUTPUT_VARIABLE REF_DIRTY COMMAND_ERROR_IS_FATAL ANY)
if(REF_DIRTY)
 message(FATAL_ERROR "Reference source must remain byte-identical and clean")
endif()
set(HAL ${REFERENCE_ROOT}/Drivers/STM32U5xx_HAL_Driver)
set(CMSIS ${REFERENCE_ROOT}/Drivers/CMSIS)
set(TINY ${REFERENCE_ROOT}/Middlewares/TinyUSB/src)
file(GLOB HAL_SOURCES ${HAL}/Src/*.c)
file(GLOB CORE_SOURCES ${REFERENCE_ROOT}/Core/Src/*.c)
add_executable(sdmmc1_poc ${CORE_SOURCES} ${HAL_SOURCES}
 ${TINY}/tusb.c ${TINY}/common/tusb_fifo.c ${TINY}/device/usbd.c ${TINY}/device/usbd_control.c
 ${TINY}/class/msc/msc_device.c ${TINY}/portable/synopsys/dwc2/dcd_dwc2.c ${TINY}/portable/synopsys/dwc2/dwc2_common.c
 ${REFERENCE_ROOT}/Middlewares/FatFs/src/ff.c ${REFERENCE_ROOT}/Middlewares/FatFs/src/diskio.c
 ${REFERENCE_ROOT}/STM32CubeIDE/Application/User/Startup/startup_stm32u575zitxq.s)
target_include_directories(sdmmc1_poc PRIVATE ${REFERENCE_ROOT}/Core/Inc ${HAL}/Inc ${HAL}/Inc/Legacy ${CMSIS}/Include ${CMSIS}/Device/ST/STM32U5xx/Include ${TINY} ${REFERENCE_ROOT}/Middlewares/FatFs/src)
target_compile_definitions(sdmmc1_poc PRIVATE USE_HAL_DRIVER STM32U575xx $<$<CONFIG:Debug>:DEBUG>)
target_compile_options(sdmmc1_poc PRIVATE -mcpu=cortex-m33 -mthumb -mfpu=fpv5-sp-d16 -mfloat-abi=hard  $<$<CONFIG:Release>:-Os> $<$<NOT:$<CONFIG:Release>>:-O0> -g3 -ffunction-sections -fdata-sections -Wall)
target_link_options(sdmmc1_poc PRIVATE -mcpu=cortex-m33 -mthumb -mfpu=fpv5-sp-d16 -mfloat-abi=hard --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/sdmmc1_poc.map -T${REFERENCE_ROOT}/STM32CubeIDE/STM32U575ZITXQ_FLASH.ld)
set_target_properties(sdmmc1_poc PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET sdmmc1_poc POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:sdmmc1_poc> ${CMAKE_CURRENT_BINARY_DIR}/sdmmc1_poc.bin COMMAND ${SIZE} $<TARGET_FILE:sdmmc1_poc>)
