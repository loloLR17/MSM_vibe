cmake_minimum_required(VERSION 3.20)
project(shield_dual_bringup LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
option(BRINGUP_HOST_TESTS "Only hardware safety gate tests" OFF)
if(BRINGUP_HOST_TESTS)
 enable_testing()
 add_executable(test_gates tests/test_gates.c)
 target_include_directories(test_gates PRIVATE src)
 target_compile_options(test_gates PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME shield_bringup_write_gates COMMAND test_gates)
 return()
endif()
if(NOT CMAKE_CROSSCOMPILING)
 message(FATAL_ERROR "Explicit ARM toolchain required")
endif()
set(STM32CUBE_U5_ROOT "" CACHE PATH "Qualified CubeU5")
set(HAL "${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver")
set(DEV "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx")
set(STREF "${CMAKE_CURRENT_SOURCE_DIR}/../iis3dwb_st_reference")
set(SDREF "${CMAKE_CURRENT_SOURCE_DIR}/../spi2_microsd_isolated")
set(COEX "${CMAKE_CURRENT_SOURCE_DIR}/../r3_coex_fifo_fir_sd")
set(FAT "${CMAKE_CURRENT_SOURCE_DIR}/../r3_durable_fat32/vendor/fatfs")
get_filename_component(FW "${CMAKE_CURRENT_SOURCE_DIR}/../../Modbus RTU/05_Firmware" ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(OBSERVED "${CMAKE_CURRENT_BINARY_DIR}/generated/stm32u5xx_hal_spi_observed.c")
execute_process(COMMAND ${Python3_EXECUTABLE} -B "${STREF}/tools/instrument_hal.py" "${HAL}/Src/stm32u5xx_hal_spi.c" "${OBSERVED}" RESULT_VARIABLE RET)
if(NOT RET EQUAL 0)
 message(FATAL_ERROR "Existing qualified HAL fingerprint refused")
endif()
set(SDK)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex)
 list(APPEND SDK "${HAL}/Src/stm32u5xx_${module}.c")
endforeach()
add_executable(shield_bringup src/main.c "${STREF}/src/platform.c" "${STREF}/src/spi_diagnostic.c" "${STREF}/vendor/st/iis3dwb_reg.c" "${COEX}/src/board_sd.c" "${COEX}/src/sd_writer.c" "${SDREF}/src/sd_spi.c" "${FW}/platform/stm32/stm32_fram_storage.c" "${FAT}/ff.c" "${OBSERVED}" ${SDK} "${DEV}/Source/Templates/gcc/startup_stm32u575xx.s" "${DEV}/Source/Templates/system_stm32u5xx.c")
target_include_directories(shield_bringup PRIVATE src "${STREF}/src" "${STREF}/vendor/st" "${SDREF}/src" "${COEX}/src" "${FAT}" "${CMAKE_CURRENT_SOURCE_DIR}/../r3_durable_fat32/src" "${FW}/include" "${FW}/platform/stm32" "${HAL}/Inc" "${DEV}/Include" "${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include")
target_compile_definitions(shield_bringup PRIVATE STM32U575xx USE_HAL_DRIVER BENCH_CORE_MHZ=160 BENCH_SPI_DIVIDER=8 BENCH_SPI_MODE3=1 BENCH_DUPLEX=1 BENCH_SECOND_SPI_OBSERVER=1 R3_DURABLE_CARD_PROFILE=1)
target_compile_options(shield_bringup PRIVATE -mcpu=cortex-m33 -mthumb -O2 -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_link_options(shield_bringup PRIVATE -mcpu=cortex-m33 -mthumb --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/shield_bringup.map" "-T${FW}/platform/stm32/linker/STM32U575ZITXQ_FLASH.ld")
set_target_properties(shield_bringup PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
add_custom_command(TARGET shield_bringup POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:shield_bringup> ${CMAKE_CURRENT_BINARY_DIR}/shield_bringup.bin)
