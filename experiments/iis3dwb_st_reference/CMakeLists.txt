cmake_minimum_required(VERSION 3.20)
project(iis3dwb_st_reference LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(BENCH_WINDOW_XYZ 0 CACHE STRING "Q streaming window XYZ target: 0, 512 or 4096")
if(NOT BENCH_WINDOW_XYZ MATCHES "^(0|512|4096)$")
 message(FATAL_ERROR "Unplanned XYZ window")
endif()
option(BENCH_OFFICIAL "Execute the official ST sensor function" OFF)
set(BENCH_DUPLEX 0 CACHE STRING "P causal factor: one duplex command+payload frame")
if(NOT BENCH_DUPLEX MATCHES "^(0|1)$")
 message(FATAL_ERROR "Unplanned transport strategy")
endif()
set(BENCH_SPI_DIVIDER 8 CACHE STRING "SPI3 divider: 8 historical, 4 or 2 comparison")
if(NOT BENCH_SPI_DIVIDER MATCHES "^(8|4|2)$")
 message(FATAL_ERROR "Unplanned SPI frequency")
endif()
set(BENCH_SPI_MODE3 0 CACHE STRING "SPI mode3 platform profile")
set(OFFICIAL_STATUS10 0 CACHE STRING "Explicit local FIFO level getter correction")
set(OFFICIAL_PACKETS 1 CACHE STRING "Bounded original example packets: 1 or 3")
if(NOT OFFICIAL_PACKETS MATCHES "^(1|3)$" OR NOT OFFICIAL_STATUS10 MATCHES "^(0|1)$" OR NOT BENCH_SPI_MODE3 MATCHES "^(0|1)$")
 message(FATAL_ERROR "Unplanned official profile")
endif()
option(BENCH_CLEANUP_ONLY "Safety image: stop sensor only, no acquisition" OFF)
option(BENCH_HOST_TESTS "Build only offline mock tests" OFF)
set(BENCH_SPI_OPT "2" CACHE STRING "HAL SPI only: 0 diagnostic baseline or 2 corrected service")
if(NOT BENCH_SPI_OPT MATCHES "^(0|2)$")
  message(FATAL_ERROR "BENCH_SPI_OPT must be 0 or 2")
endif()
set(BENCH_EXAMPLE "polling" CACHE STRING "who, polling or fifo")
set(BENCH_PROFILE "ST_STRICT" CACHE STRING "ST_STRICT or STATUS10")
if(NOT BENCH_EXAMPLE MATCHES "^(who|polling|fifo)$" OR
   NOT BENCH_PROFILE MATCHES "^(ST_STRICT|STATUS10)$")
  message(FATAL_ERROR "Unknown example/profile")
endif()
add_library(st_driver vendor/st/iis3dwb_reg.c)
target_include_directories(st_driver PUBLIC vendor/st)
set(BENCH_FIFO_BURST 128 CACHE STRING "Diagnostic FIFO payload words: 128 or 1")
set(BENCH_CAPTURE_ENTRIES 512 CACHE STRING "Bounded observation: 512 or 8 entries")
set(BENCH_FIFO_DECIMATION 8 CACHE STRING "Timestamp batching: 8 or 0 (disabled)")
set(BENCH_FIFO_BDR_LAST 0 CACHE STRING "Diagnostic ordering: set BDR after XL enable")
if(NOT BENCH_FIFO_BURST MATCHES "^(128|1)$" OR
   NOT BENCH_CAPTURE_ENTRIES MATCHES "^(512|8)$" OR
   NOT BENCH_FIFO_DECIMATION MATCHES "^(8|0)$" OR
   NOT BENCH_FIFO_BDR_LAST MATCHES "^(0|1)$")
  message(FATAL_ERROR "Refused unplanned FIFO diagnostic parameter")
endif()
add_library(bench src/bench.c)
target_compile_definitions(bench PUBLIC BENCH_FIFO_BURST=${BENCH_FIFO_BURST}
  BENCH_CAPTURE_ENTRIES=${BENCH_CAPTURE_ENTRIES}
  BENCH_FIFO_DECIMATION=${BENCH_FIFO_DECIMATION}
  BENCH_FIFO_BDR_LAST=${BENCH_FIFO_BDR_LAST})
target_include_directories(bench PUBLIC src)
target_link_libraries(bench PUBLIC st_driver)
target_compile_options(bench PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BENCH_HOST_TESTS)
  if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host tests require a native compiler")
  endif()
  enable_testing()
  add_executable(test_fifo_window tests/test_fifo_window.c src/fifo_window.c)
  target_include_directories(test_fifo_window PRIVATE src)
  target_link_libraries(test_fifo_window PRIVATE st_driver)
  target_compile_options(test_fifo_window PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME bounded_fifo_xyz_window COMMAND test_fifo_window)
  add_executable(test_spi_diagnostic tests/test_spi_diagnostic.c src/spi_diagnostic.c)
  target_include_directories(test_spi_diagnostic PRIVATE src)
  target_compile_options(test_spi_diagnostic PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME spi_preclose_diagnostic COMMAND test_spi_diagnostic)
  add_executable(test_reference tests/test_reference.c)
  target_link_libraries(test_reference PRIVATE bench)
  target_compile_options(test_reference PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME official_driver_and_bounded_examples COMMAND test_reference)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME fifo_independent_qualification COMMAND ${Python3_EXECUTABLE}
    -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fifo_qualification.py)
  add_test(NAME spi_hal_instrumentation COMMAND ${Python3_EXECUTABLE}
    -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hal_instrumentation.py)
  add_test(NAME offline_capture_and_target_guards COMMAND ${Python3_EXECUTABLE}
    -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.py)
  return()
endif()
if(NOT CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Use cmake/arm-none-eabi-gcc.cmake or BENCH_HOST_TESTS")
endif()
set(STM32CUBE_U5_ROOT "" CACHE PATH "Recursive STM32CubeU5 v1.9.0 checkout")
set(HAL ${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver)
set(DEV ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx)
set(HAL_SOURCES)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex hal_spi)
  set(source ${HAL}/Src/stm32u5xx_${module}.c)
  if(NOT EXISTS ${source})
    message(FATAL_ERROR "Missing CubeU5 source: ${source}")
  endif()
  list(APPEND HAL_SOURCES ${source})
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(SPI_ORIGINAL ${HAL}/Src/stm32u5xx_hal_spi.c)
set(SPI_OBSERVED ${CMAKE_CURRENT_BINARY_DIR}/generated/stm32u5xx_hal_spi_observed.c)
execute_process(COMMAND ${Python3_EXECUTABLE} -B
  ${CMAKE_CURRENT_SOURCE_DIR}/tools/instrument_hal.py ${SPI_ORIGINAL} ${SPI_OBSERVED}
  RESULT_VARIABLE instrumentation_result)
if(NOT instrumentation_result EQUAL 0)
  message(FATAL_ERROR "HAL source fingerprint/instrumentation refused")
endif()
list(REMOVE_ITEM HAL_SOURCES ${SPI_ORIGINAL})
set_source_files_properties(${SPI_OBSERVED} PROPERTIES COMPILE_OPTIONS "-O${BENCH_SPI_OPT}")
set(BENCH_MAIN src/main.c)
if(BENCH_CLEANUP_ONLY)
  set(BENCH_MAIN src/cleanup_main.c)
endif()
set(OFFICIAL_SOURCES)
set(WINDOW_SOURCES)
if(BENCH_WINDOW_XYZ)
 if(NOT BENCH_OFFICIAL OR BENCH_DUPLEX OR NOT BENCH_SPI_MODE3 OR NOT OFFICIAL_STATUS10)
  message(FATAL_ERROR "Q requires unchanged positive ST mode3/split transport")
 endif()
 set(WINDOW_SOURCES src/fifo_window.c)
 set_source_files_properties(src/fifo_window.c PROPERTIES COMPILE_OPTIONS "-O2")
endif()
if(BENCH_OFFICIAL)
  set(BENCH_MAIN official/main.c)
  set(OFFICIAL_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/generated/iis3dwb_fifo_ported.c)
  execute_process(COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/official/generate.py
    ${CMAKE_CURRENT_SOURCE_DIR}/vendor/st/examples/iis3dwb_fifo.c ${OFFICIAL_GENERATED}
    RESULT_VARIABLE official_generation)
  if(NOT official_generation EQUAL 0)
    message(FATAL_ERROR "Official provenance refused")
  endif()
  set(OFFICIAL_SOURCES official/port.c ${OFFICIAL_GENERATED})
endif()
add_executable(iis3dwb_reference ${BENCH_MAIN} ${OFFICIAL_SOURCES} ${WINDOW_SOURCES} src/platform.c src/spi_diagnostic.c ${SPI_OBSERVED} ${HAL_SOURCES}
  ${DEV}/Source/Templates/gcc/startup_stm32u575xx.s
  ${DEV}/Source/Templates/system_stm32u5xx.c)
target_include_directories(iis3dwb_reference PRIVATE src official ${HAL}/Inc ${DEV}/Include
  ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include)
target_compile_definitions(iis3dwb_reference PRIVATE STM32U575xx USE_HAL_DRIVER
  BENCH_WINDOW_XYZ=${BENCH_WINDOW_XYZ} BENCH_DUPLEX=${BENCH_DUPLEX}
  BENCH_SPI_DIVIDER=${BENCH_SPI_DIVIDER} BENCH_SPI_MODE3=${BENCH_SPI_MODE3} OFFICIAL_STATUS10=${OFFICIAL_STATUS10} OFFICIAL_PACKETS=${OFFICIAL_PACKETS}
  BENCH_FIFO=$<IF:$<STREQUAL:${BENCH_EXAMPLE},who>,2,$<STREQUAL:${BENCH_EXAMPLE},fifo>>
  BENCH_STATUS10=$<STREQUAL:${BENCH_PROFILE},STATUS10>)
foreach(target st_driver bench iis3dwb_reference)
  target_compile_options(${target} PRIVATE -mcpu=cortex-m33 -mthumb
    -ffunction-sections -fdata-sections)
endforeach()
target_compile_options(iis3dwb_reference PRIVATE -Wall -Wextra -Werror)
if(BENCH_OFFICIAL)
 target_link_libraries(iis3dwb_reference PRIVATE st_driver)
else()
 target_link_libraries(iis3dwb_reference PRIVATE bench)
endif()
target_link_options(iis3dwb_reference PRIVATE -mcpu=cortex-m33 -mthumb
  --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none
  -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/iis3dwb_reference.map
  -T${CMAKE_CURRENT_SOURCE_DIR}/STM32U575ZITXQ_FLASH.ld)
if(BENCH_OFFICIAL)
 target_link_options(iis3dwb_reference PRIVATE -Wl,-u,_printf_float)
endif()
set_target_properties(iis3dwb_reference PROPERTIES SUFFIX ".elf")
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET iis3dwb_reference POST_BUILD
  COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:iis3dwb_reference> ${CMAKE_CURRENT_BINARY_DIR}/iis3dwb_reference.bin
  COMMAND ${SIZE} $<TARGET_FILE:iis3dwb_reference>)
