cmake_minimum_required(VERSION 3.20)
project(sdmmc_st_reference LANGUAGES C ASM)
if(NOT CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Use cmake/arm-none-eabi-gcc.cmake")
endif()
set(STM32CUBE_U5_ROOT "" CACHE PATH "STM32CubeU5 v1.9.0 recursive checkout")
# Writes require a new build following explicit physical/media authorization.
set(BENCH_WRITE_ENABLE 0 CACHE STRING "0 readonly; 1 authorized bounded test")
set(BENCH_TEST_LBA 0 CACHE STRING "Explicit authorized sector, never sector zero")
if(NOT BENCH_WRITE_ENABLE MATCHES "^[01]$" OR NOT BENCH_TEST_LBA MATCHES "^[0-9]+$")
  message(FATAL_ERROR "Invalid media permission")
endif()
if(BENCH_WRITE_ENABLE AND BENCH_TEST_LBA LESS 1)
  message(FATAL_ERROR "Writing requires explicit nonzero test LBA")
endif()
set(HAL ${STM32CUBE_U5_ROOT}/Drivers/STM32U5xx_HAL_Driver)
set(DEV ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Device/ST/STM32U5xx)
set(SOURCES)
foreach(module hal hal_cortex hal_dma hal_flash hal_flash_ex hal_gpio hal_pwr hal_pwr_ex hal_rcc hal_rcc_ex hal_sd hal_sd_ex ll_sdmmc)
  set(source ${HAL}/Src/stm32u5xx_${module}.c)
  if(NOT EXISTS ${source})
    message(FATAL_ERROR "Missing ${source}")
  endif()
  list(APPEND SOURCES ${source})
endforeach()
option(BENCH_LOAD_ONLY "One fixed 600-second 1-bit TR2-rate load" OFF)
set(BENCH_MAIN src/main.c)
if(BENCH_LOAD_ONLY)
  if(NOT BENCH_WRITE_ENABLE)
    message(FATAL_ERROR "Load-only requires explicitly authorized write profile")
  endif()
  set(BENCH_MAIN src/load_main.c)
endif()
add_executable(sdmmc_reference ${BENCH_MAIN} src/clock.c ${SOURCES}
 ${DEV}/Source/Templates/gcc/startup_stm32u575xx.s
 ${DEV}/Source/Templates/system_stm32u5xx.c)
target_include_directories(sdmmc_reference PRIVATE src ${HAL}/Inc ${DEV}/Include ${STM32CUBE_U5_ROOT}/Drivers/CMSIS/Include)
target_compile_definitions(sdmmc_reference PRIVATE STM32U575xx USE_HAL_DRIVER BENCH_WRITE_ENABLE=${BENCH_WRITE_ENABLE} BENCH_TEST_LBA=${BENCH_TEST_LBA})
target_compile_options(sdmmc_reference PRIVATE -mcpu=cortex-m33 -mthumb -O2 -g3 -Wall -Wextra -Werror -ffunction-sections -fdata-sections)
target_link_options(sdmmc_reference PRIVATE -mcpu=cortex-m33 -mthumb --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections -Wl,--build-id=none -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/sdmmc_reference.map -T${CMAKE_CURRENT_SOURCE_DIR}/STM32U575ZITXQ_FLASH.ld)
set_target_properties(sdmmc_reference PROPERTIES SUFFIX .elf)
find_program(OBJCOPY arm-none-eabi-objcopy REQUIRED)
find_program(SIZE arm-none-eabi-size REQUIRED)
add_custom_command(TARGET sdmmc_reference POST_BUILD COMMAND ${OBJCOPY} -O binary $<TARGET_FILE:sdmmc_reference> ${CMAKE_CURRENT_BINARY_DIR}/sdmmc_reference.bin COMMAND ${SIZE} $<TARGET_FILE:sdmmc_reference>)
